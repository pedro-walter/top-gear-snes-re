@008000 reset_vec
;; RESET: init CPU, decompress the WRAM engine image (ROM -> $7E2000), upload the SPC700 program

@00812A main_restart
;; start of every game cycle: clears DP/WRAM $0000-$1EFF, then JSL 0F:806D (menus) happens before this
;; (from 00:8126/00:886F); sets up PPU registers, the WRAM NMI/IRQ stubs ($00/$04 = JML 00:EC1A / 00:EC5B),
;; calls load_track, builds the first frames and enters the race loop at jmp_008510.
@008510 race_loop
;; main race loop: frame_delta ($1E90) = elapsed frames+1; speeds ramp towards target ($46/$59 chase the cars'

@008834 race_exit
;; leaves the race: stores positions, disables NMI, then JSL 0F:806D with A=1 (results screens) and jumps back to main_restart
